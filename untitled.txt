    <script>

        const display = document.getElementById("display");

        const tombol = document.querySelectorAll(".tombol button");

        tombol.forEach(button => {

            button.addEventListener("click", function() {

                let nilai = this.getAttribute("data-value");

                // Tombol C
                if (nilai === "C") {
                    display.value = "";
                }

                // Tombol DEL
                else if (nilai === "DEL") {
                    display.value = display.value.slice(0, -1);
                }

                // Tombol =
                else if (nilai === "=") {

                    try {

                        let ekspresi = display.value;

                        if (!/^[0-9+\-*/. ]+$/.test(ekspresi)) {
                            throw new Error("Input salah");
                        }

                        let hasil = Function(
                            '"use strict"; return (' + ekspresi + ')'
                        )();

                        if (!Number.isFinite(hasil)) {
                            throw new Error("Hasil salah");
                        }

                        display.value = hasil;

                    } catch (error) {
                        display.value = "Error";
                    }

                }

                // Tombol angka dan operator
                else {

                    if (display.value === "Error") {
                        display.value = "";
                    }

                    display.value += nilai;

                }

            });

        });

    </script>
